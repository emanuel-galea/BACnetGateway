int dataToBinary(char * pText, uint8_t * pData) {
	char ch;
	int value;
	int dataLength = 0;
	while (*pText != 0) {
		value = 0;
		ch = *pText++;
		if ((ch >= '0') && (ch <= '9')) {
			value = ch - '0';
		} else if ((ch >= 'A') && (ch <= 'F')) {
			value = (ch - 'A') + 10;
		} else if ((ch >= 'a') && (ch <= 'f')) {
			value = (ch - 'a') + 10;
		} else {
			continue;
		}
		if (pText != 0) {
			ch = *pText++;
			value = value << 4;
			if ((ch >= '0') && (ch <= '9')) {
				ch = ch - '0';
			} else if ((ch >= 'A') && (ch <= 'F')) {
				ch = (ch - 'A') + 10;
			} else if ((ch >= 'a') && (ch <= 'f')) {
				ch = (ch - 'a') + 10;
			}
			value |= ch & 0xF;
			pData[dataLength++] = value;
		}
	}
	return dataLength;
}

char * writeFile(char * fileName, char * buffer) {

	FILE * pFile;
	pFile = fopen(fileName, "w");
	if (pFile == NULL) {
		return NULL;
	}
	fputs(buffer, pFile);
	fclose(pFile);
	return buffer;
}

uint8_t * readFile(char * fileName, int * pLength) {
	long lSize;
	long result;
	uint8_t * buffer;
	FILE * pFile;
	pFile = fopen(fileName, "r");
	if (pFile == NULL) {
		//		fprintf(stderr, "File error: %s \n", filePath);
		return NULL;
	}

	// obtain file size:
	fseek(pFile, 0, SEEK_END);
	lSize = ftell(pFile);
	rewind(pFile);

	buffer = (uint8_t *) malloc(sizeof(char) * lSize);
	if (buffer == NULL) {
		fputs("Memory error", stderr);
		return NULL;
	}

	// copy the file into the buffer:
	result = fread(buffer, 1, lSize, pFile);
	if (result != lSize) {
		fputs("Reading error", stderr);
		free(buffer);
		return NULL;
	}
	*pLength = lSize;
	fclose(pFile);
	//	free(buffer);
	return buffer;
}

int openSerialInterface(int baudrateIndex, int parity) {
	char message[100];
	int retValue, fd;

	serialPort.portName = &serialPortName[0];
	if ((fd = open(serialPort.portName, O_RDWR | O_NOCTTY | O_NDELAY)) < 0) {
		sprintf(&message[0], "ERROR: Failed to open port %s error %s", serialPortName, strerror(errno));
		printf(message);
		return -1;
	}
	serialPort.portFD = fd;
	serialPort.baudrateIndex = baudrateIndex;
	/* save off original settings */
	bzero(&tty_state, sizeof(struct termios));
	//	tty_state.c_lflag = 0; //ICANON;
	tty_state.c_iflag |= INPCK; // Enable input parity checking.
	tty_state.c_lflag &= ~(ICANON | ECHO | ECHOE | ISIG);
	tty_state.c_cflag |= (PARENB | CLOCAL | B2400); // Enable Parity  ignore modem lines
	switch (parity) {
	case PARITY_NONE:
		tty_state.c_cflag &= ~(PARENB); //Disbale parity
		break;
	case PARITY_EVEN:
		tty_state.c_cflag &= ~(PARODD); // Enable Parity EVEN
		break;
	case PARITY_ODD:
		tty_state.c_cflag |= PARODD; // Enable Parity ODD
		break;
	}
	tty_state.c_cflag &= ~(CSTOPB | CSIZE); // EVEN Parity, only one stop bit
	tty_state.c_cflag |= CS8; // 8 bit
	tty_state.c_cflag |= CREAD; // enable receiver
	//	tty_state.c_cflag |= CRTSCTS; // enable RTS

	tty_state.c_cc[VMIN] = 0;
	tty_state.c_cc[VTIME] = 0; // 500 milli seconds
	cfsetispeed(&tty_state, baudrateInterface[baudrateIndex]);
	cfsetospeed(&tty_state, baudrateInterface[baudrateIndex]);

	tcflush(fd, TCIFLUSH);
	if ((retValue = tcsetattr(fd, TCSAFLUSH, &tty_state)) < 0) {
		sprintf(&message[0], "Setting Serial Port settings %s tcsetattr failed with error: %s", serialPortName,
				strerror(errno));
		return -3;
	}
	ioctl(fd, TCFLSH, TCIOFLUSH);
	return fd;
}

/**
 *
 * @return 0 = nothing received, LT 0  means error, GT 0 lenght of data received
 *
 */
int readSerialPort(SerialPort * pSerialPort) {
	fd_set rset;
	int retValue;
	int i;
	struct timeval tv;
	int recIndex = 0;
	int timeout;
	//	timeout = baudrateDelay[pSerialPort->baudrateIndex];
	timeout = 2000;
	while (1) {
		if (timeout >= 1000) {
			tv.tv_sec = timeout / 1000;
			tv.tv_usec = (timeout % 1000) * 1000;
		} else {
			tv.tv_sec = 0;
			tv.tv_usec = timeout * 1000;
		}
		FD_ZERO(&rset);
		FD_SET (pSerialPort->portFD, &rset);
		retValue = select(pSerialPort->portFD + 1, &rset, NULL, NULL, &tv);
		if (retValue == 0) {
			//			printf("Timeout\n");
			retValue = recIndex;
			break;
		} else if (retValue < 0) {
			fprintf(stderr, "Select Error: %s\n", strerror(errno));
			break;
		} else {
			if (FD_ISSET(pSerialPort->portFD, &rset)) {
				/* data coming from device */
				retValue = read(pSerialPort->portFD, &recBuffer[recIndex], REC_BUFFER_LENGTH - recIndex);
				recIndex += retValue;
				//				printf("Data recieved length=%d  -  ", retValue);
				if (debug) {
					for (i = 0; i < recIndex; i++) {
						printf("%2X ", recBuffer[i]);
					}
					printf("\n ");
				}
				//				fprintf(stderr, "data recieved %d", retValue);

				timeout = 150;
				return recIndex;
			}
			//			tv.tv_sec = 0;
			//			tv.tv_usec = 150 * 1000; // 150 milli
			//			usleep(10000);		// 10 milli
			if (debug) {
				printf("Data Received %d ", retValue);
			}
		}

	} // while
	return retValue;
}
